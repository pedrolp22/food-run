function(subway_library target alias)
    add_library(${target} STATIC ${ARGN})
    add_library(subway::${alias} ALIAS ${target})
    target_compile_features(${target} PUBLIC cxx_std_20)
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

if(SUBWAY_BUILD_PLATFORM)
    # Prefer a system SDL2 installation, but provide a self-contained fallback
    # so a fresh CLion toolchain can configure this project without manual
    # SDL2_DIR/CMAKE_PREFIX_PATH setup.
    find_package(SDL2 CONFIG QUIET)
    if(NOT TARGET SDL2::SDL2 AND NOT TARGET SDL2::SDL2-static AND NOT TARGET SDL2::SDL2-shared)
        include(FetchContent)
        FetchContent_Declare(SDL2
            URL "https://github.com/libsdl-org/SDL/releases/download/release-2.30.12/SDL2-2.30.12.zip"
            URL_HASH "SHA256=aa2808d0f2dc6b383c6689bf6d166e2de62db4d58be989e4b052acb31df0fba3"
        )
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
        set(SDL_TEST OFF CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(SDL2)
    endif()
    subway_library(subway_platform platform src/engine/platform/Window.cpp)
    target_link_libraries(subway_platform PRIVATE subway_glad)
    target_compile_definitions(subway_platform PUBLIC SDL_MAIN_HANDLED)
    if(WIN32)
        set(sdl_candidates SDL2::SDL2-static SDL2::SDL2-shared SDL2::SDL2)
    else()
        set(sdl_candidates SDL2::SDL2 SDL2::SDL2-static SDL2::SDL2-shared)
    endif()
    foreach(candidate IN LISTS sdl_candidates)
        if(TARGET ${candidate})
            set(SUBWAY_SDL_TARGET ${candidate})
            break()
        endif()
    endforeach()
    if(NOT SUBWAY_SDL_TARGET)
        message(FATAL_ERROR "SDL2 did not provide a usable library target")
    endif()
    target_link_libraries(subway_platform PUBLIC ${SUBWAY_SDL_TARGET})
endif()

# Imported shared SDL packages need their DLL beside every Windows executable.
function(subway_stage_sdl target)
    if(WIN32 AND SUBWAY_SDL_TARGET)
        get_target_property(sdl_type ${SUBWAY_SDL_TARGET} TYPE)
        if(sdl_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET ${target} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:${SUBWAY_SDL_TARGET}>" "$<TARGET_FILE_DIR:${target}>"
                VERBATIM)
        endif()
    endif()
endfunction()
